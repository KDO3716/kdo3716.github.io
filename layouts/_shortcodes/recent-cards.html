{{- /* 최신 글 카드 (1:1 썸네일 + 아래 제목/요약) 사용법: {{< recent-cards section="project" count="3" requireImage="true" >}} - section: 글을 가져올 섹션 (content 아래 폴더 이름) - count: 보여줄 글 개수 (기본 3) - requireImage: "true"면 이 섹션의 글 중 이미지가 없는 글이 있을 때 빌드를 멈춤 썸네일 순서: 글 맨 위의 thumbnail → 본문의 첫 이미지 → /images/No Image.png */ -}} {{- $section := .Get "section" -}} {{- $count := .Get "count" | default 3 | int -}} {{- $requireImage := eq (.Get "requireImage") "true" -}} {{- $fallback := "images/No%20Image.png" | relURL -}} {{- $all := (where site.RegularPages "Section" $section).ByDate.Reverse -}} {{- /* 각 글의 썸네일 주소 찾기 */ -}} {{- $thumbs := dict -}} {{- range $all -}} {{- $thumb := .Params.thumbnail | default "" -}} {{- if not $thumb -}} {{- with findRESubmatch `!\[[^\]]*\]\(\s*]+)|]*?\ssrc=["']([^"']+)["']` .RawContent 1 -}} {{- $m := index . 0 -}} {{- $thumb = or (index $m 1) (index $m 2) -}} {{- end -}} {{- end -}} {{- if $thumb -}} {{- if not (or (hasPrefix $thumb "/") (hasPrefix $thumb "http")) -}} {{- /* 글 폴더 안에 있는 이미지 (예: image.png) */ -}} {{- with .Resources.Get $thumb -}}{{- $thumb = .RelPermalink -}}{{- end -}} {{- else if hasPrefix $thumb "/" -}} {{- $thumb = relURL (strings.TrimPrefix "/" $thumb) -}} {{- end -}} {{- else if $requireImage -}} {{- errorf "%q 글에 이미지가 없습니다. 글 맨 위에 thumbnail을 지정하거나 본문에 이미지를 1개 이상 넣어주세요." .File.Path -}} {{- end -}} {{- $thumbs = merge $thumbs (dict .RelPermalink ($thumb | default $fallback)) -}} {{- end -}}
