{{- /* 최신 글 카드 (1:1 썸네일 + 아래 제목/요약) 사용법: {{< recent-cards section="project" count="3" requireImage="true" >}} - section: 글을 가져올 섹션 (content 아래 폴더 이름) - count: 보여줄 글 개수 (기본 3) - requireImage: "true"면 이 섹션의 글 중 이미지가 없는 글이 있을 때 경고를 출력 (카드에는 No_Image.webp 표시) 카드 모양은 layouts/_partials/card-grid.html 에서 수정합니다. */ -}} {{- $section := .Get "section" -}} {{- $count := .Get "count" | default 3 | int -}} {{- $requireImage := eq (.Get "requireImage") "true" -}} {{- $pages := first $count (where site.RegularPages "Section" $section).ByDate.Reverse -}} {{- partial "card-grid.html" (dict "pages" $pages "requireImage" $requireImage) -}}
